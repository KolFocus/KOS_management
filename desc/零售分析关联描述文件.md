2:投广账号总数 COUNT(IF([投广笔记数] IS NOT NULL, [所属用户], NULL))
3:发布笔记数 SUM(IF([投广笔记数] IS NOT NULL, [发布笔记数], NULL))
4:投广笔记数 sum(投广笔记数)
5:笔记总互动数 SUM(IF([投广笔记数] IS NOT NULL, [总互动数], NULL))
6:投广带来的互动量 SUM(IF([投广笔记数] IS NOT NULL, [互动量], NULL))
7:私信进线数 SUM(IF([投广笔记数] IS NOT NULL, [私信进线人数], NULL))
8:私信开口数 SUM(IF([投广笔记数] IS NOT NULL, [私信开口人数], NULL))
9:企微留资数 sum(企微留资数)
10:企微招募效率 SUM(IF([投广笔记数] IS NOT NULL,[企微留资数], NULL))*1.0/SUM(IF([投广笔记数] IS NOT NULL, [私信进线人数], NULL))
11:企微留资成本 SUM([笔记投广消耗])*1.0/SUM(IF([投广笔记数] IS NOT NULL,[企微留资数], NULL))
12:期间成交量 sum(小红书成单)
