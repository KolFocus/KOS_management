#### 2.3 API请求封装 (api/request.js)
```javascript
import axios from 'axios'
import { ElMessage } from 'element-plus'

const request = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL,
  timeout: 10000
})

// 请求拦截器
request.interceptors.request.use(
  config => {
    const token = localStorage.getItem('token')
    if (token) {
      config.headers.Authorization = `Bearer ${token}`
    }
    return config
  },
  error => {
    return Promise.reject(error)
  }
)

// 响应拦截器
request.interceptors.response.use(
  response => {
    const { code, message, data } = response.data
    if (code === '200') {
      return data
    } else {
      ElMessage.error(message)
      return Promise.reject(new Error(message))
    }
  },
  error => {
    ElMessage.error('网络错误，请稍后重试')
    return Promise.reject(error)
  }
)

export default request
```

### 2. 核心文件说明

#### 2.1 统一返回结果 (common/Result.java)
```java
@Data
@AllArgsConstructor
@NoArgsConstructor
public class Result<T> {
    private String code;
    private String message;
    private T data;
    private String timestamp;
    
    public static <T> Result<T> success(T data) {
        return new Result<>("200", "操作成功", data, 
            LocalDateTime.now().format(DateTimeFormatter.ISO_LOCAL_DATE_TIME));
    }
    
    public static <T> Result<T> error(String code, String message) {
        return new Result<>(code, message, null, 
            LocalDateTime.now().format(DateTimeFormatter.ISO_LOCAL_DATE_TIME));
    }
}
```

#### 2.5 KOS列表控制器 (controller/KosListController.java)
```java
@RestController
@RequestMapping("/api/kos")
@Api(tags = "KOS列表管理")
@Slf4j
public class KosListController {
    
    @Autowired
    private KosListService kosListService;
    
    @GetMapping("/list")
    @ApiOperation("获取KOS列表")
    public Result<PageResult<KosListVO>> getKosList(
        @ApiParam("品牌ID") @RequestParam(required = false) String brandId,
        @ApiParam("渠道") @RequestParam(required = false) String channel,
        @ApiParam("状态") @RequestParam(required = false) Integer status,
        @ApiParam("页码") @RequestParam(defaultValue = "1") Integer pageNum,
        @ApiParam("页大小") @RequestParam(defaultValue = "20") Integer pageSize
    ) {
        try {
            KosListDTO query = new KosListDTO();
            query.setBrandId(brandId);
            query.setChannel(channel);
            query.setStatus(status);
            query.setPageNum(pageNum);
            query.setPageSize(pageSize);
            
            PageResult<KosListVO> result = kosListService.getKosList(query);
            return Result.success(result);
        } catch (Exception e) {
            log.error("获取KOS列表失败", e);
            return Result.error("500", "获取KOS列表失败");
        }
    }
}
```
